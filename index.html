data = transpose(datos)

media = (arr, f) => arr.length ? arr.reduce((s, d) => s + f(d), 0) / arr.length : NaN

// Formato numérico: muestra "–" si no hay datos con los filtros elegidos
fmt = (x, dec = 1) => Number.isFinite(x) ? x.toFixed(dec) : "–"

// Clases del histograma de horas: límites fijos calculados sobre el dataset completo
horas_min = d3.min(data, d => d.study_hours_per_week)
horas_max = d3.max(data, d => d.study_hours_per_week)
n_clases  = 8
amplitud  = (horas_max - horas_min) / n_clases
cortes    = d3.range(1, n_clases).map(i => horas_min + i * amplitud)

// Tabla de frecuencias sobre los datos filtrados
tabla_frec = {
  const bins = d3.bin()
    .domain([horas_min, horas_max])
    .thresholds(cortes)
    .value(d => d.study_hours_per_week)(filtrado);
  const n = filtrado.length;
  let acum = 0;
  return bins.map((b, i) => {
    acum += b.length;
    const cierre = i === bins.length - 1 ? "]" : ")";
    return {
      intervalo: `[${b.x0.toFixed(1)}, ${b.x1.toFixed(1)}${cierre}`,
      frec_abs:  b.length,
      frec_rel:  n ? b.length / n : 0,
      frec_porc: n ? 100 * b.length / n : 0,
      frec_acum: acum
    };
  });
}

// Asimetría (misma fórmula que e1071::skewness, type = 3) sobre los datos filtrados
asim = {
  const n = filtrado.length;
  if (n < 3) return NaN;
  const m  = d3.mean(filtrado, d => d.study_hours_per_week);
  const m2 = d3.mean(filtrado, d => (d.study_hours_per_week - m) ** 2);
  const m3 = d3.mean(filtrado, d => (d.study_hours_per_week - m) ** 3);
  return (m3 / m2 ** 1.5) * ((n - 1) / n) ** 1.5;
}

interpretacion = !Number.isFinite(asim)
  ? "No hay datos suficientes con los filtros elegidos."
  : Math.abs(asim) < 0.5
    ? "Al ser un valor cercano a 0, la distribución es aproximadamente simétrica."
    : asim > 0
      ? "Al ser un valor positivo, la distribución tiene asimetría a la derecha (cola hacia más horas de estudio)."
      : "Al ser un valor negativo, la distribución tiene asimetría a la izquierda (cola hacia menos horas de estudio)."
