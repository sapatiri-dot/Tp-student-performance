  {label: "Acceso a internet", value: ["Yes", "No"]})

viewof extra = Inputs.checkbox(["Yes", "No"],
  {label: "Extraescolares", value: ["Yes", "No"]})

viewof educ = Inputs.checkbox(niveles,
  {label: "Educación de los padres", value: niveles})

viewof asist_min = Inputs.range(rango_a,
  {label: "Asistencia mínima (%)", step: 1, value: rango_a[0]})

filtrado = data.filter(d =>
  genero.includes(d.gender) &&
  internet.includes(d.internet_access) &&
  extra.includes(d.extracurricular) &&
  educ.includes(d.parent_education) &&
  d.attendance_rate >= asist_min
